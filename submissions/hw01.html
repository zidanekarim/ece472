19class Parameters:  # avoided calling this simple BasisExpansion since it's not really the same

26class NNXLinearModel(nnx.Module):
27    """A Flax NNX module for a linear regression model."""
28
29    def __init__(self, *, rngs: nnx.Rngs, num_features: int):
30        self.num_features = num_features
31        key = rngs.params()
32        self.w = nnx.Param(jax.random.normal(key, (self.num_features, 1)))
33        self.b = nnx.Param(jnp.zeros((1, 1)))
34
35    def __call__(self, x: jax.Array) -> jax.Array:
36        """Predicts the output for a given input."""
37        return jnp.squeeze(x @ self.w.value + self.b.value, axis=-1)
38
39    @property
40    def model(self) -> LinearModel:
41        """Returns the underlying simple linear model."""
42        return LinearModel(
43            weights=np.array(self.w.value).reshape([self.num_features]),
44            bias=float(np.array(self.b.value).squeeze()),
45        )
46
47
48class NNXBasisExpansion(nnx.Module):
49    def __init__(self, *, rngs: nnx.Rngs, num_features: int, M: int):
50        self.num_features = num_features
51        self.M = M
52        key = rngs.params()
53        # src: https://flax.readthedocs.io/en/stable/api_reference/flax.nnx/variables.html
54        # param required for learnable & automatic diff
55        self.mu = nnx.Param(
56            jnp.linspace(0.0, 1.0, num=self.M)
57        )  # had to ask about min/max, but makes sense since we are sampling x from (0,1)
58        # log sig here cuz theyll die out otherwise, thanks Josh Miao
59        self.sig = nnx.Param(
60            jnp.ones((self.M,)) * jnp.log((1.0 / self.M))
61        )  # 0.16? let M=6, 1/M
62        # sig uses ones so no key needed
63
64    def __call__(self, x: jax.Array):
65        """Predicts the output for a given input."""  # copied from NNXLinearModel
66        #  to ensure broadcasting we need to reshape x here
67        if x.ndim == 1:
68            x = x[
69                :, jnp.newaxis
70            ]  # adds singleton axis (equivalent to None, but dummy axis essentially)
71        # restore sig when we are computing phi
72        return jnp.exp(-((x - self.mu.value) ** 2) / jnp.exp(self.sig.value) ** 2)
73
74
75class NNXRegressionModel(nnx.Module):
76    def __init__(self, *, rngs: nnx.Rngs, num_features: int, M: int):
77        self.M = M
78        self.basis = NNXBasisExpansion(rngs=rngs, num_features=num_features, M=self.M)
79        self.linear = NNXLinearModel(rngs=rngs, num_features=self.M)
80
81    def __call__(self, x: jax.Array) -> jax.Array:
82        phi_x = self.basis(x)
83        return self.linear(
84            phi_x
85        )  # we can pass phi_x in place of x as according to given functional form
86        # and we need to read from this regression rather than the straight linear
87
88    @property
89    def model(self) -> Parameters:
90        return Parameters(
91            mu=np.array(self.basis.mu.value),
92            sigma=np.array(self.basis.sig.value),
93            weights=np.array(self.linear.w.value).reshape([self.linear.num_features]),
94            bias=float(np.array(self.linear.b.value).squeeze()),
95        )

14@nnx.jit
15def train_step(
16    model: NNXRegressionModel,
17    optimizer: nnx.Optimizer,
18    x: jnp.ndarray,
19    y: jnp.ndarray,  # changed here to NNXRegressionModel
20):
21    """Performs a single training step."""
22
23    def loss_fn(model: NNXRegressionModel):
24        y_hat = model(x)
25        return 0.5 * jnp.mean((y_hat - y) ** 2)
26
27    loss, grads = nnx.value_and_grad(loss_fn)(model)
28    optimizer.update(model, grads)  # In-place update of model parameters
29    return loss
30
31
32def train(
33    model: NNXRegressionModel,
34    optimizer: nnx.Optimizer,
35    data: Data,
36    settings: TrainingSettings,
37    np_rng: np.random.Generator,
38) -> None:
39    """Train the model using SGD."""
40    log.info("Starting training", **settings.model_dump())
41    bar = trange(settings.num_iters)
42    for i in bar:
43        x_np, y_np = data.get_batch(np_rng, settings.batch_size)
44        x, y = jnp.asarray(x_np), jnp.asarray(y_np)
45
46        loss = train_step(model, optimizer, x, y)
47
48        bar.set_description(f"Loss @ {i} => {loss:.6f}")
49        bar.refresh()
50    log.info("Training finished")

17    for key, value in event_dict.items():
