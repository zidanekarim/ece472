 43class Classifier(nnx.Module):
 44    def __init__(
 45        self,
 46        input_channels: int,
 47        layer_channels: list[int],
 48        kernel_sizes: list[tuple[int, int]],
 49        strides: list[int],
 50        num_classes: int,
 51        *,
 52        rngs,
 53    ):
 54        self.conv_layers = nnx.List([])
 55        self.input_channels = input_channels
 56        self.linear_layer = NNXLinearModel(
 57            rngs, in_features=layer_channels[-1], out_features=num_classes
 58        )
 59
 60        for output_channel, kernel_size, stride in zip(
 61            layer_channels, kernel_sizes, strides
 62        ):
 63            self.conv_layers.append(
 64                Conv2d(
 65                    in_features=self.input_channels,
 66                    out_features=output_channel,
 67                    kernel_size=kernel_size,
 68                    strides=stride,
 69                    padding="SAME",
 70                    use_bias=True,
 71                    rngs=rngs,
 72                )  # padding="SAME" to keep the output size the same as input size
 73            )
 74            self.input_channels = output_channel  # similar to resnet where we set the input channels to the output channels of the previous layer
 75
 76    def __call__(self, x: jnp.ndarray) -> jnp.ndarray:
 77        for conv_layer in self.conv_layers:
 78            x = nnx.relu(
 79                conv_layer(x)
 80            )  # convultional layers commonly use ReLu as the activation function
 81        x = jnp.mean(
 82            x, axis=(1, 2)
 83        )  # global average pooling so model is invariant to input resolution
 84        x = self.linear_layer(x)
 85        return x

 88class NNXLinearModel(nnx.Module):
 89    """A Flax NNX module for a linear regression model, with multi-dimension support added to solve MLP"""
 90
 91    def __init__(self, rngs: nnx.Rngs, in_features: int, out_features: int):
 92        key = rngs.params()
 93        self.in_features = in_features  # bc of MLP
 94        self.out_features = out_features
 95        std = jnp.sqrt(
 96            2.0 / in_features
 97        )  # prevents vanishing gradients, switched to He initialization
 98        self.w = nnx.Param(jax.random.normal(key, (in_features, out_features)) * std)
 99        self.b = nnx.Param(jnp.zeros((1, out_features)))  # broadcasts for output
100
101    def __call__(self, x: jax.Array) -> jax.Array:
102        """Predicts the output for a given input."""
103        return x @ self.w.value + self.b.value  # squeeze gone to maintain hidden layers

14@nnx.jit
15def train_step(model, optimizer: nnx.Optimizer, x: jnp.ndarray, y: jnp.ndarray):
16    """Performs a single training step."""
17
18    def loss_fn(model):
19        logits = model(x)
20        return jnp.mean(optax.softmax_cross_entropy_with_integer_labels(logits, y))
21
22    loss, grads = nnx.value_and_grad(loss_fn)(model)
23    optimizer.update(
24        model, grads
25    )  # In-place update of model parameters, same as second assignment
26    return loss
27
28
29def train(
30    model,
31    optimizer: nnx.Optimizer,
32    data: Data,
33    settings: TrainingSettings,
34    np_rng: np.random.Generator,
35) -> None:
36    log.info("Starting training", **settings.model_dump())
37    bar = trange(settings.num_iters)
38    for i in bar:
39        x_np, y_np = data.get_batch(np_rng, settings.batch_size)
40        x, y = jnp.asarray(x_np), jnp.asarray(y_np)
41
42        loss = train_step(model, optimizer, x, y)
43        bar.set_description(f"Loss @ {i} => {loss:.6f}")
44        bar.refresh()
45    log.info("Training finished")

52@nnx.jit
53def eval_step(model: nnx.Module, x: jnp.ndarray, y: jnp.ndarray):
54    logits = model(x)
55    predicted_classes = jnp.argmax(logits, axis=-1)
56    correct_count = jnp.sum(predicted_classes == y)
57    return correct_count, x.shape[0]

60def evaluate(model: nnx.Module, test_dataset, batch_size: int = 256) -> float:
61    total_correct = 0
62    total_samples = 0
63
64    num_samples = len(test_dataset)
65    for start_idx in range(0, num_samples, batch_size):
66        end_idx = min(start_idx + batch_size, num_samples)
67        batch = test_dataset[start_idx:end_idx]
68
69        x = jnp.asarray(batch["image"], dtype=jnp.float32)
70        if x.ndim == 3:
71            x = jnp.expand_dims(x, axis=-1)
72
73        y = jnp.asarray(batch["label"])
74
75        correct, count = eval_step(model, x, y)
76        total_correct += int(correct)
77        total_samples += int(count)
78
79    accuracy = total_correct / total_samples
80    return accuracy

16def custom_serializer_processor(logger, method_name, event_dict):
17    for key, value in event_dict.items():
18        # Handle JAX arrays in addition to TF tensors
19        if hasattr(value, "numpy"):  # Covers TF tensors
20            value = value.numpy()
21        if isinstance(value, jax.Array):
22            value = np.array(value)

24            value = value.item() if value.size == 1 else value.tolist()
25        if isinstance(value, float):
26            value = FormattedFloat(value)
27        if isinstance(value, Path):
